Quake / Carmack Lightstyle Pattern
// Carmack's classic fluorescent flicker
const char *flicker = "mmamammmmammamamaaamammma";

int brightness(int t) {
    char c = flicker[t % strlen(flicker)];
    return (c == 'm') ? 100 : 0; // m = full, a = dark
}
